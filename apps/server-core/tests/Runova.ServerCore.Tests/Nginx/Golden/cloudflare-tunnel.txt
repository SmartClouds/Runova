==> http.conf <==
# Managed by Runova. Each apply writes a new release, so change this in the app instead.

# WebSocket requests ask the upstream to switch protocols; other requests close as usual.
map $http_upgrade $runova_connection_upgrade {
    default upgrade;
    "" close;
}

# Requests for names no site claims get no answer at all, and TLS handshakes for them
# are refused before any certificate is shown.
server {
    listen 80 default_server;
    listen [::]:80 default_server;
    listen 443 ssl default_server;
    listen [::]:443 ssl default_server;
    server_name _;
    ssl_reject_handshake on;
    return 444;
}

# Through the Cloudflare Tunnel the visitor's scheme comes in X-Forwarded-Proto; cloudflared always sets it.
map $http_x_forwarded_proto $runova_tunnel_scheme {
    default https;
    http http;
}

map $http_x_forwarded_proto $runova_tunnel_port {
    default 443;
    http 80;
}

# The tunnel's listener: names no site claims get no answer.
server {
    listen 127.0.0.2:18080 default_server;
    server_name _;
    return 444;
}

include /etc/nginx/runova/releases/1/sites/tunneled.conf;

==> sites/tunneled.conf <==
# Site tunneled: tunneled.test

# Plain HTTP answers ACME challenges and sends everything else to HTTPS.
server {
    listen 80;
    listen [::]:80;
    server_name tunneled.test;
    server_tokens off;
    access_log /var/log/nginx/runova-tunneled.access.log;
    error_log /var/log/nginx/runova-tunneled.error.log;

    location ^~ /.well-known/acme-challenge/ {
        root /var/www/runova/acme;
        default_type text/plain;
        try_files $uri =404;
        allow all;
        auth_basic off;
    }

    location / {
        return 301 https://$host$request_uri;
    }
}

server {
    listen 443 ssl;
    listen [::]:443 ssl;
    http2 on;
    server_name tunneled.test;
    server_tokens off;
    root /var/www/runova/sites/tunneled;
    access_log /var/log/nginx/runova-tunneled.access.log;
    error_log /var/log/nginx/runova-tunneled.error.log;

    ssl_certificate /etc/runova-test/tls/sites.crt;
    ssl_certificate_key /etc/runova-test/tls/sites.key;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305;
    ssl_session_cache shared:runova_tls:10m;
    ssl_session_timeout 1d;
    ssl_session_tickets off;
    ssl_stapling off;

    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        # The visitor's own address replaces any X-Forwarded-For they sent, so it cannot be forged.
        proxy_set_header X-Forwarded-For $remote_addr;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
    }
}

# Through the Cloudflare Tunnel: only cloudflared on this machine reaches this listener.
server {
    listen 127.0.0.2:18080;
    server_name tunneled.test;
    server_tokens off;
    root /var/www/runova/sites/tunneled;
    access_log /var/log/nginx/runova-tunneled.access.log;
    error_log /var/log/nginx/runova-tunneled.error.log;

    include /etc/nginx/runova/releases/1/cloudflare/tunnel-real-ip.conf;

    location ^~ /.well-known/acme-challenge/ {
        root /var/www/runova/acme;
        default_type text/plain;
        try_files $uri =404;
        allow all;
        auth_basic off;
    }

    if ($runova_tunnel_scheme = "http") {
        return 301 https://$host$request_uri;
    }

    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        # The visitor's own address replaces any X-Forwarded-For they sent, so it cannot be forged.
        proxy_set_header X-Forwarded-For $remote_addr;
        proxy_set_header X-Forwarded-Proto $runova_tunnel_scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $runova_tunnel_port;
    }
}
